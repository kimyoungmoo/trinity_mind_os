### HSL → 숫자 인코딩

```typescript
function hslToNumber(h: number, s: number, l: number): number {
  const hBits = Math.round(h) & 0x1FF;  // 9비트
  const sBits = Math.round(s) & 0x7F;   // 7비트
  const lBits = Math.round(l) & 0x7F;   // 7비트
  
  return (hBits << 14) | (sBits << 7) | lBits;
}

// 예시
hslToNumber(345, 95, 55) → 5636407
```

### HSLA (투명도 포함)

```typescript
function hslaToNumber(h: number, s: number, l: number, a: number): number {
  const hBits = Math.round(h) & 0x1FF;        // 9비트
  const sBits = Math.round(s) & 0x7F;         // 7비트
  const lBits = Math.round(l) & 0x7F;         // 7비트
  const aBits = Math.round(a * 255) & 0xFF;   // 8비트
  
  return (hBits << 22) | (sBits << 15) | (lBits << 8) | aBits;
}

// W축과 투명도 연결
const W_ALPHA = {

  3: 0.4,   // 의식계
  4: 0.2,   // 초의식계
  5: 0.0,   // 본질계 (투명)
};
```
